/-
GeodesicDistance.lean. Chapter 0 section 0.10 and chapter 3 section 3.3, geodesic distance.

The geodesic distance between two nodes of a graph is the length of the shortest path
between them. It is zero from a node to itself, symmetric, and on a connected graph obeys
the triangle inequality. The geodesic-rank reader of chapter 3 reads only the ordering of
these distances, so any strictly increasing transform of them leaves what it reads
unchanged, which is the radius-and-angle argument in one line. The file takes the
distance from Mathlib's simple graphs.
-/
import Mathlib
import DataMiningAsObservation.RankFaithful

namespace DMO.GeodesicDistance

variable {V : Type*} (G : SimpleGraph V)

/-- **A node is at distance zero from itself.** -/
theorem dist_self (v : V) : G.dist v v = 0 := SimpleGraph.dist_self

/-- **The distance is symmetric.** -/
theorem dist_comm (u v : V) : G.dist u v = G.dist v u := SimpleGraph.dist_comm

/-- **The triangle inequality** on a connected graph. -/
theorem dist_triangle (hG : G.Connected) (u v w : V) : G.dist u w ≤ G.dist u v + G.dist v w :=
  hG.dist_triangle

/-- **Adjacent nodes are at distance one.** -/
theorem dist_adj {u v : V} (h : G.Adj u v) : G.dist u v = 1 := SimpleGraph.dist_eq_one_iff_adj.2 h

/-- **The geodesic-rank reader is blind to a strictly increasing transform** of the distances.
Its nearest neighbour is the same under the transformed distance. -/
theorem geodesic_rank_invariant [Fintype V] (g : ℝ → ℝ) (hg : StrictMono g) (S : Finset V)
    (q c : V) (hc : ∀ v ∈ S, (G.dist q c : ℝ) ≤ G.dist q v) :
    ∀ v ∈ S, g (G.dist q c) ≤ g (G.dist q v) :=
  DMO.RankFaithful.nearest_eq g (DMO.RankFaithful.isRankFaithful_of_strictMono g hg) S
    (fun v => (G.dist q v : ℝ)) c hc

end DMO.GeodesicDistance
