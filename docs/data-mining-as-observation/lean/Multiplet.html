/-
Multiplet.lean. Chapter 0 section 0.15 and chapter 9 section 9.2, the multiplet.

A multiplet is a group of eigenvalues that are equal or nearly so. The cycle on `n` nodes,
the discrete circle, has Laplacian eigenvalues `2 − 2 cos(2πk/n)`, and the value at `k`
equals the value at `n − k`, so every eigenvalue other than the ones at `k = 0` and, for
even `n`, at `k = n/2` appears twice. That pairing is the multiplet the recognizer of
chapter 9 reads as the circle's signature.
-/
import Mathlib

namespace DMO.Multiplet

open Real

/-- The `k`-th Laplacian eigenvalue of the cycle on `n` nodes. -/
noncomputable def cycleEig (n k : ℕ) : ℝ := 2 - 2 * Real.cos (2 * π * k / n)

/-- **The constant mode has eigenvalue zero.** -/
theorem cycleEig_zero (n : ℕ) : cycleEig n 0 = 0 := by
  simp [cycleEig]

/-- **Every eigenvalue of the cycle is nonnegative.** -/
theorem cycleEig_nonneg (n k : ℕ) : 0 ≤ cycleEig n k := by
  unfold cycleEig
  linarith [Real.cos_le_one (2 * π * k / n)]

/-- **Every eigenvalue of the cycle is at most four.** -/
theorem cycleEig_le_four (n k : ℕ) : cycleEig n k ≤ 4 := by
  unfold cycleEig
  linarith [Real.neg_one_le_cos (2 * π * k / n)]

/-- **Every eigenvalue of the cycle is paired, `k` with `n − k`.** -/
theorem cycleEig_pair (n k : ℕ) (hk : k ≤ n) (hn : 0 < n) :
    cycleEig n (n - k) = cycleEig n k := by
  unfold cycleEig
  have hn' : (n : ℝ) ≠ 0 := Nat.cast_ne_zero.mpr hn.ne'
  have : (2 * π * ((n - k : ℕ) : ℝ) / n) = 2 * π - 2 * π * k / n := by
    rw [Nat.cast_sub hk]
    field_simp
  rw [this, Real.cos_two_pi_sub]

/-- **The two members of a pair are distinct modes unless `2k = n`.** -/
theorem pair_distinct (n k : ℕ) (hk : k ≤ n) (h : 2 * k ≠ n) : n - k ≠ k := by
  omega

end DMO.Multiplet
